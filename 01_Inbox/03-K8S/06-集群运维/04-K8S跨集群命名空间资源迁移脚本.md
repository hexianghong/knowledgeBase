# K8S 跨集群命名空间资源迁移脚本

本文档提供了一个用于将 Kubernetes 某个命名空间下的所有核心资源（包括 Deployment、Service、StatefulSet、Ingress、PVC、PV、ConfigMap、Secret 等）迁移到另一个集群的 Shell 脚本。

为了保证生产环境执行的安全性，该脚本已开启了严格的错误检测模式，并对用户输入和边界情况进行了容错处理。

## 脚本依赖
该脚本依赖以下工具：
- `kubectl`: 用于与 Kubernetes API 交互。
- `jq`: 用于处理和清理 JSON 格式的 Kubernetes 资源清单。

## 迁移脚本 `migrate-ns.sh`

创建一个文件 `migrate-ns.sh`，并赋予执行权限 `chmod +x migrate-ns.sh`。

```bash
#!/usr/bin/env bash
# 
# 用途: 迁移 Kubernetes 命名空间及其内部资源到新集群
# 依赖: kubectl, jq

# ================= 安全配置 =================
# 开启严格模式：
# -e: 遇到命令执行失败即退出
# -u: 遇到未定义的变量即退出
# -o pipefail: 管道命令中任何一个失败，整个管道即刻失败
set -euo pipefail

# ================= 全局变量 =================
COMMAND=""
NAMESPACE=""
BACKUP_DIR=""
KUBECONFIG_PATH=""

# 要迁移的命名空间级别资源类型
# 增加了 HPA, PDB, NetworkPolicy, RBAC 等更多核心资源
RESOURCES=(
  "configmap"
  "secret"
  "serviceaccount"
  "role"
  "rolebinding"
  "pvc"
  "service"
  "deployment"
  "daemonset"
  "statefulset"
  "ingress"
  "cronjob"
  "job"
  "horizontalpodautoscaler"
  "poddisruptionbudget"
  "networkpolicy"
  "limitrange"
  "resourcequota"
)

function usage() {
    cat <<EOF
使用方法: $0 <export|import> [参数]

导出命令 (export):
  $0 export -n <namespace> [-d <backup_dir>] [-k <kubeconfig>]
    -n  指定要导出的源命名空间 (必填)
    -d  指定导出存放目录 (可选，默认: ./k8s_migration_<namespace>_YYYYMMDDHHMMSS)
    -k  指定使用的 kubeconfig 路径 (可选，默认使用系统当前配置)

导入命令 (import):
  $0 import -n <namespace> -d <backup_dir> [-k <kubeconfig>]
    -n  指定要导入到的目标命名空间 (必填)
    -d  指定包含导出数据的目录路径 (必填)
    -k  指定使用的 kubeconfig 路径 (可选，默认使用系统当前配置)

EOF
    exit 1
}

if [[ $# -lt 1 ]]; then
    usage
fi

COMMAND=$1
shift

while getopts "n:d:k:" opt; do
    case ${opt} in
        n ) NAMESPACE=$OPTARG ;;
        d ) BACKUP_DIR=$OPTARG ;;
        k ) KUBECONFIG_PATH=$OPTARG ;;
        \? ) usage ;;
    esac
done

# ================= 依赖检查与输入校验 =================
if ! command -v jq &> /dev/null; then
    echo "[错误] 缺少 jq 命令，请先安装 jq (例如: yum install jq 或 apt-get install jq)。"
    exit 1
fi

if [[ -z "${NAMESPACE}" ]]; then
    echo "[错误] 必须使用 -n 指定命名空间。"
    usage
fi

# 校验命名空间合法性 (防止路径穿越和非法的 K8s 名称)
if  ! "${NAMESPACE}" =~ ^a-z0-9?$ ; then
    echo "[错误] 命名空间名称不合法: ${NAMESPACE} (必须符合 RFC 1123 规范)"
    exit 1
fi

# 动态配置 KUBECONFIG
if [[ -n "${KUBECONFIG_PATH}" ]]; then
    if [[ ! -f "${KUBECONFIG_PATH}" ]]; then
        echo "[错误] 指定的 kubeconfig 文件不存在: ${KUBECONFIG_PATH}"
        exit 1
    fi
    export KUBECONFIG="${KUBECONFIG_PATH}"
fi

# ================= 导出逻辑 =================
function do_export() {
    if [[ -z "${BACKUP_DIR}" ]]; then
        BACKUP_DIR="./k8s_migration_${NAMESPACE}_$(date +%Y%m%d%H%M%S)"
    fi
    
    if ! kubectl cluster-info &> /dev/null; then
        echo "[错误] 无法连接到源集群，请检查 kubeconfig 配置及集群网络。"
        exit 1
    fi

    echo ">>> 开始从集群 [${NAMESPACE}] 命名空间导出资源到 [${BACKUP_DIR}]..."
    mkdir -p "${BACKUP_DIR}"

    # PV 是集群级别的资源。通过读取 PVC 找对应 PV
    echo ">> [1/2] 导出关联的 PersistentVolume (PV)..."
    mkdir -p "${BACKUP_DIR}/pv"
    
    # 提取非空的 PV 列表（换行符分割）
    local PV_LIST
    PV_LIST=$(kubectl -n "${NAMESPACE}" get pvc -o jsonpath='{range .items[*]}{.spec.volumeName}{"\n"}{end}' | grep -v '^$' || true)

    if [[ -n "${PV_LIST}" ]]; then
        # 使用 while read 防止 PV 过多导致 for 循环超出 Bash ARG_MAX 限制
        while read -r PV; do
            echo "   - 导出 PV: ${PV}"
            # 安全地读取并过滤 json，解除其与旧 PVC 的强绑定
            # || true 防止个别 PV 获取失败导致脚本直接中断
            kubectl get pv "${PV}" -o json 2>/dev/null | jq '
                del(.metadata.resourceVersion, .metadata.uid, .metadata.creationTimestamp, .metadata.managedFields, .metadata.ownerReferences, .metadata.annotations["pv.kubernetes.io/bind-completed"], .metadata.annotations["pv.kubernetes.io/bound-by-controller"], .status, .spec.claimRef)
            ' > "${BACKUP_DIR}/pv/${PV}.json" || true
        done <<< "${PV_LIST}"
    else
        echo "   - 未发现关联的 PV"
    fi

    # 导出并清理命名空间资源
    echo ">> [2/2] 导出命名空间级别资源..."
    for RES in "${RESOURCES[@]}"; do
        mkdir -p "${BACKUP_DIR}/${RES}"
        
        local ITEMS
        # 核心修复 1: 过滤掉拥有 ownerReferences 的资源（例如 CronJob 自动生成的 Job），防止导入时发生级联冲突
        # 核心修复 2: 使用 while read 替代 for in 避免 ARG_MAX 越界
        ITEMS=$(kubectl -n "${NAMESPACE}" get "${RES}" -o json 2>/dev/null | jq -r '.items[]? | select(.metadata.ownerReferences == null) | .metadata.name' || true)
        
        if [[ -z "${ITEMS}" ]]; then
            continue
        fi
        
        echo "   - 导出资源类型: ${RES}"
        while read -r ITEM; do
            [[ -z "${ITEM}" ]] && continue
            
            # 过滤掉 ServiceAccount 自动生成的系统 token
            if [[ "${RES}" == "secret" ]] && [[ "${ITEM}" =~ ^(default-token|builder-token)-.*$ ]]; then
                continue
            fi
            
            # 使用 jq 安全处理元数据
            # 核心修复 3: 针对 Service 的 ClusterIP，不能暴力删除！如果值为 "None" (Headless Service)，删除它会导致新集群为其分配普通 IP 从而破坏 Headless 逻辑
            kubectl -n "${NAMESPACE}" get "${RES}" "${ITEM}" -o json 2>/dev/null | jq '
                del(.metadata.resourceVersion, .metadata.uid, .metadata.creationTimestamp, .metadata.managedFields, .metadata.ownerReferences, .metadata.generateName, .status) |
                (if .kind == "Service" and .spec.clusterIP != "None" then del(.spec.clusterIP, .spec.clusterIPs) else . end) |
                (if .kind == "Service" then del(.spec.ports[].nodePort) else . end) |
                (if .kind == "PersistentVolumeClaim" then del(.metadata.annotations["pv.kubernetes.io/bind-completed"], .metadata.annotations["pv.kubernetes.io/bound-by-controller"]) else . end)
            ' > "${BACKUP_DIR}/${RES}/${ITEM}.json" || true
            
        done <<< "${ITEMS}"
    done

    echo ">>> 资源导出和清理完成！"
    echo ">>> 导出内容保存在目录: ${BACKUP_DIR}"
    echo ">>> 您可以将此目录打包并传输到目标集群所在的主机进行导入。"
}

# ================= 导入逻辑 =================
function do_import() {
    if [[ -z "${BACKUP_DIR}" ]]; then
        echo "[错误] 导入命令必须使用 -d 指定包含导出数据的目录路径。"
        usage
    fi

    if [[ ! -d "${BACKUP_DIR}" ]]; then
        echo "[错误] 目录 ${BACKUP_DIR} 不存在。"
        exit 1
    fi

    if ! kubectl cluster-info &> /dev/null; then
        echo "[错误] 无法连接到目标集群，请检查 kubeconfig 配置及集群网络。"
        exit 1
    fi

    echo ">>> 开始将 ${BACKUP_DIR} 中的资源导入到集群的 [${NAMESPACE}] 命名空间..."

    # 检查目标命名空间是否存在
    if ! kubectl get ns "${NAMESPACE}" >/dev/null 2>&1; then
        echo ">> 目标命名空间 ${NAMESPACE} 不存在，正在创建..."
        kubectl create ns "${NAMESPACE}"
    fi

    # 启用 nullglob，防止文件夹为空时 *.json 被解析为字面量而报错
    shopt -s nullglob

    # 导入 PV (集群级别)
    local pv_files=("${BACKUP_DIR}/pv/"*.json)
    if [[ ${#pv_files[@]} -gt 0 ]]; then
        echo ">> 导入 PV..."
        for FILE in "${pv_files[@]}"; do
            echo "   - Apply $(basename "${FILE}")"
            kubectl apply -f "${FILE}"
        done
    fi

    # 按顺序导入命名空间级别资源
    for RES in "${RESOURCES[@]}"; do
        local res_files=("${BACKUP_DIR}/${RES}/"*.json)
        if [[ ${#res_files[@]} -gt 0 ]]; then
            echo ">> 导入资源类型: ${RES}..."
            for FILE in "${res_files[@]}"; do
                echo "   - Apply $(basename "${FILE}")"
                # 安全替换 namespace 并应用 (使用 jq --arg 防注入)，避免在导出阶段写死 namespace
                jq --arg ns "${NAMESPACE}" '.metadata.namespace = $ns' "${FILE}" | kubectl apply -n "${NAMESPACE}" -f -
            done
        fi
    done
    
    # 恢复 nullglob 设置
    shopt -u nullglob

    echo ">>> 跨集群迁移导入任务完成！请检查目标集群服务状态。"
}

# ================= 路由分发 =================
case ${COMMAND} in
    export)
        do_export
        ;;
    import)
        do_import
        ;;
    *)
        echo "[错误] 未知命令: ${COMMAND}"
        usage
        ;;
esac
```

## 注意事项与建议

1. **执行方式变更**：
   - 导出数据时，登录到旧集群主机，执行 `export` 动作。
   - 然后将生成的目录压缩打包（如 `tar -czvf backup.tar.gz k8s_migration_xxx`），通过 `scp` 等方式传输到新集群主机。
   - 在新集群主机解压，并执行 `import` 动作恢复资源。

2. **PV 数据持久化同步（非常重要）**
   - 此脚本只会迁移 K8s 的 API 对象定义（如 YAML / JSON 文件中的规格配置）。**它不会复制底层物理存储卷中的数据。**
   - 如果你需要迁移数据内容，请确保新集群可以访问到原集群相同的存储后端（例如相同的 NFS Server 路径或 Ceph 集群）。如果不互通，你需要在运行脚本前，利用 `rsync` 或者专用的数据备份工具（如 **Velero**）对真实数据进行同步。

3. **存储类 (StorageClass)**
   - 如果新旧集群的 `StorageClass` 名称不同，需要先在新集群中创建同名的 `StorageClass`，或者在执行导入前通过文本替换将 JSON 文件中的 `storageClassName` 改为目标集群可用的。

4. **NodePort 冲突**
   - 脚本中刻意使用 jq 删除了 `Service` 的 `nodePort` 字段 (`del(.spec.ports[].nodePort)`)，目的是让新集群自动随机分配新的端口，以防止和新集群的已有端口产生冲突。如果业务强依赖固定端口，需手动修改或将该逻辑注释掉。
